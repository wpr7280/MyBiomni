```typescript
import { useEffect, useRef, useState } from 'react';
import { io, Socket } from 'socket.io-client';

export const useWebSocket = (conversationId: string) => {
  const [messages, setMessages] = useState<Message[]>([]);
  const [executionSteps, setExecutionSteps] = useState<ExecutionStep[]>([]);
  const socketRef = useRef<Socket | null>(null);

  useEffect(() => {
    socketRef.current = io(WS_URL, {
      auth: { token: getAuthToken() }
    });

    socketRef.current.on('message', (message: Message) => {
      setMessages(prev => [...prev, message]);
    });

    socketRef.current.on('execution_step', (step: ExecutionStep) => {
      setExecutionSteps(prev => [...prev, step]);
    });

    return () => {
      socketRef.current?.disconnect();
    };
  }, [conversationId]);

  return { messages, executionSteps, socket: socketRef.current };
};
```

```java
        log.warn("  Username: admin");

    private String generateSecurePassword() {
        // 生成16位强密码：大小写字母、数字、特殊字符
        String upperCase = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
        String lowerCase = "abcdefghijklmnopqrstuvwxyz";
        String digits = "0123456789";
        String special = "!@#$%^&*";
        String allChars = upperCase + lowerCase + digits + special;
        
        SecureRandom random = new SecureRandom();
        StringBuilder password = new StringBuilder(16);
        
        // 确保至少包含每种字符
        password.append(upperCase.charAt(random.nextInt(upperCase.length())));
        password.append(lowerCase.charAt(random.nextInt(lowerCase.length())));
        password.append(digits.charAt(random.nextInt(digits.length())));
        password.append(special.charAt(random.nextInt(special.length())));
        
        // 填充剩余字符
        for (int i = 4; i < 16; i++) {
            password.append(allChars.charAt(random.nextInt(allChars.length())));
        }
        
        // 打乱顺序
        List<Character> chars = password.chars()
            .mapToObj(c -> (char) c)
            .collect(Collectors.toList());
        Collections.shuffle(chars, random);
        
        return chars.stream()
            .map(String::valueOf)
            .collect(Collectors.joining());
    }
```

```java
@Override
protected void doFilterInternal(HttpServletRequest request, 
                                HttpServletResponse response, 
                                FilterChain filterChain) throws ServletException, IOException {
    String token = extractToken(request);
    
    if (token != null && jwtTokenProvider.validateToken(token)) {
        Authentication auth = jwtTokenProvider.getAuthentication(token);
        SecurityContextHolder.getContext().setAuthentication(auth);
        
        // 检查是否需要强制修改密码
        CustomUserDetails userDetails = (CustomUserDetails) auth.getPrincipal();
        if (userDetails.isForcePasswordChange()) {
            String requestUri = request.getRequestURI();
            // 只允许访问修改密码接口
            if (!requestUri.equals("/api/admin/auth/change-password") 
                && !requestUri.equals("/api/admin/auth/logout")) {

                    Result.error(403, "首次登录需要修改密码")
                ));
                return;
            }
        }
    }
    
    filterChain.doFilter(request, response);
}
```

```java
// security/JwtTokenProvider.java
@Component
public class JwtTokenProvider {
    
    @Value("${jwt.secret}")
    private String jwtSecret;
    
    @Value("${jwt.expiration}")
    private long jwtExpiration;
    
    public String generateToken(UserDetails userDetails) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", ((CustomUserDetails) userDetails).getUserId());
        claims.put("userType", ((CustomUserDetails) userDetails).getUserType());
        
        return Jwts.builder()
                .setClaims(claims)
                .setSubject(userDetails.getUsername())
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + jwtExpiration))
                .signWith(SignatureAlgorithm.HS512, jwtSecret)
                .compact();
    }
    
    public boolean validateToken(String token) {
        try {
            Jwts.parser().setSigningKey(jwtSecret).parseClaimsJws(token);
            return true;
        } catch (JwtException | IllegalArgumentException e) {
            return false;
        }
    }
    
    public String getUsernameFromToken(String token) {
        Claims claims = Jwts.parser()
                .setSigningKey(jwtSecret)
                .parseClaimsJws(token)
                .getBody();
        return claims.getSubject();
    }
}
```

```python
from fastapi import FastAPI, WebSocket, BackgroundTasks
from biomni.agent import A1

app = FastAPI()

    await websocket.accept()
    
    # 注册回调
    callback = WebSocketCallback(websocket)
    
    try:
        while True:
            data = await websocket.receive_text()
            # 处理客户端消息
    except WebSocketDisconnect:
        # 清理连接
        pass

async def execute_agent_task(
    task_id: str,
    user_id: int,
    conversation_id: str,
    query: str,
    config: dict
):
    """执行 Agent 任务"""
    try:
        # 获取或创建 Agent 实例
        agent = get_or_create_agent(user_id, config)
        
        # 设置回调
        callback = DatabaseCallback(conversation_id)
        
        # 执行任务
        result = agent.go(query, callback=callback)
        
        # 保存结果
        save_result(conversation_id, result)
        
        # 通知完成
        notify_completion(conversation_id, result)
        
    except Exception as e:
        # 错误处理
        handle_error(conversation_id, e)

def get_or_create_agent(user_id: int, config: dict) -> A1:
    """获取或创建 Agent 实例"""
    if user_id not in agent_pool:
        agent_pool[user_id] = A1(
            path=config.get('data_path', './data'),
            llm=config.get('llm', 'claude-sonnet-4-5'),
            temperature=config.get('temperature', 0.7),
            commercial_mode=config.get('commercial_mode', False)
        )
    return agent_pool[user_id]
```

```python
        self._save_step({
            'conversation_id': self.conversation_id,
            'step_type': 'tool_call',
            'tool_name': tool_name,
            'tool_input': json.dumps(tool_input),
            'status': 'running'
        })
    
    def on_tool_result(self, tool_name: str, tool_output: any):
        """工具结果"""
        self._update_step({
            'tool_output': str(tool_output),
            'status': 'success',
            'completed_at': datetime.now()
        })
    
    def on_step_end(self, step_name: str, result: any):
        """步骤结束"""
        self._update_step({
            'status': 'success',
            'completed_at': datetime.now()
        })
    
    def on_error(self, error: Exception):
        """错误处理"""
        self._update_step({
            'status': 'failed',
            'error_message': str(error),
            'completed_at': datetime.now()
        })
    
    def _save_step(self, data: dict):
        """保存步骤到数据库"""
        requests.post(
            f'{self.backend_url}/api/internal/execution-steps',
            json=data
        )
    
    def _update_step(self, data: dict):
        """更新步骤"""
        requests.put(
            f'{self.backend_url}/api/internal/execution-steps/latest',
            json=data
        )
```

#### 4.2.3 消息队列集成

```python
# queue/consumer.py
import pika
import json
from core.agent_executor import execute_agent_task

class TaskConsumer:
    """任务消费者"""
    
    def __init__(self):
        self.connection = pika.BlockingConnection(
            pika.ConnectionParameters(host='rabbitmq')
        )
        self.channel = self.connection.channel()
        self.channel.queue_declare(queue='agent_tasks', durable=True)
    
    def start(self):
        """开始消费"""
        self.channel.basic_qos(prefetch_count=1)
        self.channel.basic_consume(
            queue='agent_tasks',
            on_message_callback=self.callback
        )
        
        print('Waiting for tasks...')
        self.channel.start_consuming()
    
    def callback(self, ch, method, properties, body):
        """处理任务"""
        try:
            task = json.loads(body)
            print(f'Received task: {task["task_id"]}')
            
            # 执行任务
            execute_agent_task(
                task_id=task['task_id'],
                user_id=task['user_id'],
                conversation_id=task['conversation_id'],
                query=task['query'],
                config=task['config']
            )
            
            # 确认消息
            ch.basic_ack(delivery_tag=method.delivery_tag)
            
        except Exception as e:
            print(f'Error processing task: {e}')
            # 拒绝消息，重新入队
            ch.basic_nack(delivery_tag=method.delivery_tag, requeue=True)

if __name__ == '__main__':
    consumer = TaskConsumer()
    consumer.start()
```

```java
// service/UserService.java
@Service
public class UserService {
    
    @Autowired
    private UserRepository userRepository;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private static final String USER_CACHE_KEY = "user:";
    private static final long USER_CACHE_TTL = 30 * 60; // 30分钟
    
    public User getUserById(Long userId) {
        // 先查缓存
        String cacheKey = USER_CACHE_KEY + userId;
        User user = (User) redisTemplate.opsForValue().get(cacheKey);
        
        if (user != null) {
            return user;
        }
        
        // 缓存未命中，查数据库
        user = userRepository.findById(userId)
                .orElseThrow(() -> new NotFoundException("用户不存在"));
        
        // 写入缓存
        redisTemplate.opsForValue().set(cacheKey, user, USER_CACHE_TTL, TimeUnit.SECONDS);
        
        return user;
    }
    
    public void updateUser(User user) {
        userRepository.save(user);
        
        // 删除缓存
        String cacheKey = USER_CACHE_KEY + user.getId();
        redisTemplate.delete(cacheKey);
    }
}
```

### 6.3 缓存分类

| 缓存类型 | Key 格式 | TTL | 说明 |

| 系统配置 | `config:{configKey}` | 1小时 | 系统配置 |
| 配额信息 | `quota:{userId}` | 5分钟 | 用户配额 |
| 文档列表 | `knowhow:list:{page}` | 10分钟 | 文档列表 |
| 会话 Token | `session:{token}` | 7天 | 用户会话 |

```java
@Service
public class FileStorageService {

            PutObjectArgs.builder()
                .bucket(bucketName)
                .object(objectName)

                .contentType(file.getContentType())
                .build()
        );

            GetObjectArgs.builder()
                .bucket(bucketName)
                .object(objectName)
                .build()
        );
    }

        return minioClient.getPresignedObjectUrl(
            GetPresignedObjectUrlArgs.builder()
                .bucket(bucketName)
                .object(objectName)
                .expiry(expirySeconds)
                .build()
        );
    }
}
```

### 11.2 敏感数据加密

```java
// util/EncryptionUtil.java
@Component
public class EncryptionUtil {
    
    @Value("${encryption.secret-key}")
    private String secretKey;
    
    public String encrypt(String plainText) throws Exception {
        SecretKeySpec key = new SecretKeySpec(secretKey.getBytes(), "AES");
        Cipher cipher = Cipher.getInstance("AES/ECB/PKCS5Padding");
        cipher.init(Cipher.ENCRYPT_MODE, key);
        byte[] encrypted = cipher.doFinal(plainText.getBytes());
        return Base64.getEncoder().encodeToString(encrypted);
    }
    
    public String decrypt(String encryptedText) throws Exception {
        SecretKeySpec key = new SecretKeySpec(secretKey.getBytes(), "AES");
        Cipher cipher = Cipher.getInstance("AES/ECB/PKCS5Padding");
        cipher.init(Cipher.DECRYPT_MODE, key);
        byte[] decrypted = cipher.doFinal(Base64.getDecoder().decode(encryptedText));
        return new String(decrypted);
    }
}
```
