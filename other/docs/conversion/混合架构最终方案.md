## 3. 核心流程

### 3.1 用户登录流程

```
1. 用户输入用户名密码
   ↓
2. React 前端 → Spring Boot: POST /api/auth/login
   ↓
3. Spring Boot 验证用户名密码
   ↓
4. Spring Boot 生成 JWT Token
   ↓
5. 返回 Token 给前端
   ↓
6. 前端保存 Token 到 localStorage
```

### 3.2 对话执行流程

```
1. 用户发送消息
   ↓
2. React 前端 → Spring Boot: POST /api/conversations/{id}/messages
   - 携带 JWT Token
   - 检查配额
   - 保存用户消息
   ↓
3. Spring Boot 返回消息 ID
   ↓
4. React 前端 → Python Agent: WebSocket 连接
   - 携带 JWT Token（由 Spring Boot 生成）
   - Python 验证 Token（共享密钥）
   ↓
5. 前端通过 WebSocket 发送消息
   ↓
6. Python Agent 执行 WarpHelix Agent
   - 实时推送执行步骤
   - 采集 Token 消耗
   - 采集执行时长
   ↓
7. Python Agent 保存执行步骤到数据库
   ↓
8. Python Agent 保存 Assistant 消息到数据库
   ↓
9. Python Agent 调用 Spring Boot API 更新配额
   - POST /api/internal/quota/update
   ↓
10. Python Agent 通过 WebSocket 推送完成消息
```

### 3.3 配额更新流程

```
1. Python Agent 执行完成
   ↓
2. Python Agent → Spring Boot: POST /api/internal/quota/update
   {
     "user_id": 123,
     "tokens": 1500,
     "conversation_count": 1,
     "duration_ms": 5000
   }
   ↓
3. Spring Boot 更新配额使用记录
   ↓
4. 返回成功
```

### 4.1 API 客户端配置

```typescript
// api/client.ts
import axios from 'axios';

// Spring Boot API 客户端
export const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_URL, // http://localhost:8083
  timeout: 30000,
});

// 请求拦截器：添加 Token
apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截器：处理 401
apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);
```

### 4.2 WebSocket Hook（连接 Python）

```typescript
// hooks/useWebSocket.ts
import { useEffect, useState, useRef } from 'react';

export function useWebSocket(conversationId: string) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [executionSteps, setExecutionSteps] = useState<ExecutionStep[]>([]);
  const [isExecuting, setIsExecuting] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);

  useEffect(() => {
    const token = localStorage.getItem('token'); // Spring Boot 生成的 Token
    
    // 连接 Python Agent 服务
    const ws = new WebSocket(
      `${import.meta.env.VITE_WS_URL}/ws/chat/${conversationId}?token=${token}`
    );

    ws.onopen = () => {
      console.log('WebSocket connected to Python Agent');
    };

    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      switch (data.type) {
        case 'execution_start':
          setIsExecuting(true);
          setExecutionSteps([]);
          break;
        
        case 'execution_step':
          setExecutionSteps(prev => {
            const existing = prev.findIndex(s => s.id === data.step.id);
            if (existing >= 0) {
              const newSteps = [...prev];
              newSteps[existing] = data.step;
              return newSteps;
            }
            return [...prev, data.step];
          });
          break;
        
        case 'execution_complete':
          setIsExecuting(false);
          setMessages(prev => [...prev, data.message]);
          break;
        
        case 'execution_error':
          setIsExecuting(false);
          message.error(data.message);
          break;
      }
    };

    wsRef.current = ws;

    return () => {
      ws.close();
    };
  }, [conversationId]);

  const sendMessage = async (content: string) => {
    // 1. 先调用 Spring Boot API 保存消息和检查配额
    try {
      await apiClient.post(`/api/conversations/${conversationId}/messages`, {
        content
      });
      
      // 2. 通过 WebSocket 发送给 Python Agent 执行
      if (wsRef.current?.readyState === WebSocket.OPEN) {
        wsRef.current.send(JSON.stringify({
          type: 'send_message',
          content
        }));
        
        // 立即添加用户消息到界面
        setMessages(prev => [...prev, {
          role: 'user',
          content,
          createdAt: new Date().toISOString()
        }]);
      }
    } catch (error) {
      message.error('发送失败');
    }
  };

  return {
    messages,
    executionSteps,
    isExecuting,
    sendMessage
  };
}
```

```java
// controller/ConversationController.java
@RestController
@RequestMapping("/api/conversations")
public class ConversationController {
    
    @Autowired
    private ConversationService conversationService;
    
    @Autowired
    private QuotaService quotaService;
    
    /**
     * 获取对话列表
     */
    @GetMapping
    public Result<Page<ConversationDTO>> getConversations(
        @RequestParam(defaultValue = "1") int page,
        @RequestParam(defaultValue = "20") int size
    ) {
        Long userId = getCurrentUserId();
        Page<Conversation> conversations = conversationService
            .getUserConversations(userId, page, size);
        return Result.success(conversations);
    }
    
    /**
     * 创建新对话
     */
    @PostMapping
    public Result<ConversationDTO> createConversation(
        @RequestBody CreateConversationRequest request
    ) {
        Long userId = getCurrentUserId();
        Conversation conversation = conversationService
            .createConversation(userId, request.getTitle());
        return Result.success(conversation);
    }
    
    /**
     * 发送消息（检查配额并保存）
     */
    @PostMapping("/{conversationId}/messages")
    public Result<MessageDTO> sendMessage(
        @PathVariable Long conversationId,
        @RequestBody SendMessageRequest request
    ) {
        Long userId = getCurrentUserId();
        
        // 1. 检查配额
        if (!quotaService.checkQuota(userId)) {
            throw new BusinessException("配额不足");
        }
        
        // 2. 验证对话权限
        conversationService.checkAccess(conversationId, userId);
        
        // 3. 保存用户消息
        Message userMessage = messageService.saveUserMessage(
            conversationId,
            request.getContent()
        );
        
        // 4. 返回消息（前端会通过 WebSocket 连接 Python 执行）
        return Result.success(userMessage);
    }
    
    /**
     * 获取历史消息
     */
    @GetMapping("/{conversationId}/messages")
    public Result<List<MessageDTO>> getMessages(
        @PathVariable Long conversationId,
        @RequestParam(defaultValue = "1") int page,
        @RequestParam(defaultValue = "50") int size
    ) {
        Long userId = getCurrentUserId();
        conversationService.checkAccess(conversationId, userId);
        
        List<Message> messages = messageService
            .getMessages(conversationId, page, size);
        return Result.success(messages);
    }
}
```

## 6. Python Agent 实现

### 6.1 WebSocket 服务

```python
# api/websocket.py
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Query
from core.security import verify_jwt_token  # 验证 Spring Boot 的 Token
from services.agent_service import AgentService
import httpx

router = APIRouter()

# WebSocket 连接管理器
class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[str, WebSocket] = {}
    
    async def connect(self, conversation_id: str, websocket: WebSocket):
        await websocket.accept()
        self.active_connections[conversation_id] = websocket
    
    def disconnect(self, conversation_id: str):
        if conversation_id in self.active_connections:
            del self.active_connections[conversation_id]
    
    async def send_message(self, conversation_id: str, message: dict):
        if conversation_id in self.active_connections:
            try:
                await self.active_connections[conversation_id].send_json(message)
            except Exception as e:
                print(f"发送消息失败: {e}")

manager = ConnectionManager()

@router.websocket("/ws/chat/{conversation_id}")
async def websocket_endpoint(
    websocket: WebSocket,
    conversation_id: str,
    token: str = Query(...)  # Spring Boot 生成的 Token
):
    """WebSocket 连接处理"""
    
    # 1. 验证 Token（Spring Boot 生成的）
    payload = verify_jwt_token(token)  # 使用共享密钥验证
    if not payload:
        await websocket.close(code=1008, reason="Invalid token")
        return
    
    user_id = int(payload.get("sub"))
    
    # 2. 验证权限（调用 Spring Boot API）
    has_access = await check_conversation_access(
        conversation_id, 
        user_id, 
        token
    )
    if not has_access:
        await websocket.close(code=1008, reason="Access denied")
        return
    
    # 3. 建立连接
    await manager.connect(conversation_id, websocket)
    
    try:
        while True:
            # 接收前端消息
            data = await websocket.receive_json()
            
            if data['type'] == 'send_message':
                # 执行 Agent
                await handle_agent_execution(
                    conversation_id=conversation_id,
                    user_id=user_id,
                    content=data['content'],
                    token=token,
                    manager=manager
                )
    
    except WebSocketDisconnect:
        manager.disconnect(conversation_id)
    except Exception as e:
        print(f"WebSocket error: {e}")
        manager.disconnect(conversation_id)

async def check_conversation_access(
    conversation_id: str, 
    user_id: int, 
    token: str
) -> bool:
    """调用 Spring Boot API 验证权限"""
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(
                f"{SPRING_BOOT_URL}/api/conversations/{conversation_id}",
                headers={"Authorization": f"Bearer {token}"}
            )
            return response.status_code == 200
    except:
        return False

async def handle_agent_execution(
    conversation_id: str,
    user_id: int,
    content: str,
    token: str,
    manager: ConnectionManager
):
    """执行 Agent 并推送结果"""
    
    try:
        # 1. 通知开始执行
        await manager.send_message(conversation_id, {
            'type': 'execution_start'
        })
        
        # 2. 执行 Agent
        result = await AgentService.execute_agent(
            conversation_id=int(conversation_id),
            user_id=user_id,
            query=content,
            manager=manager
        )
        
        # 3. 保存 Assistant 消息到数据库
        assistant_message = await save_assistant_message(
            conversation_id=int(conversation_id),
            content=result['content'],
            input_tokens=result['input_tokens'],
            output_tokens=result['output_tokens']
        )
        
        # 4. 调用 Spring Boot API 更新配额
        await update_quota_via_springboot(
            user_id=user_id,
            tokens=result['total_tokens'],
            conversation_count=1,
            duration_ms=result['duration_ms'],
            token=token
        )
        
        # 5. 通知前端完成
        await manager.send_message(conversation_id, {
            'type': 'execution_complete',
            'message': {
                'id': assistant_message.id,
                'role': 'assistant',
                'content': assistant_message.content,
                'created_at': assistant_message.created_at.isoformat()
            }
        })
        
    except Exception as e:
        await manager.send_message(conversation_id, {
            'type': 'execution_error',
            'message': str(e)
        })

async def update_quota_via_springboot(
    user_id: int,
    tokens: int,
    conversation_count: int,
    duration_ms: int,
    token: str
):
    """调用 Spring Boot API 更新配额"""
    async with httpx.AsyncClient() as client:
        await client.post(
            f"{SPRING_BOOT_URL}/api/internal/quota/update",
            json={
                "user_id": user_id,
                "tokens": tokens,
                "conversation_count": conversation_count,
                "duration_ms": duration_ms
            },
            headers={
                "X-Internal-Secret": INTERNAL_SECRET  # 内部调用密钥
            }
        )
```

### 6.2 JWT Token 验证（共享密钥）

```python
# core/security.py
from jose import jwt, JWTError
from core.config import settings

def verify_jwt_token(token: str) -> dict:
    """验证 Spring Boot 生成的 JWT Token"""
    try:
        # 使用与 Spring Boot 相同的密钥和算法
        payload = jwt.decode(
            token,
            settings.JWT_SECRET_KEY,  # 与 Spring Boot 共享
            algorithms=[settings.JWT_ALGORITHM]  # HS256
        )
        return payload
    except JWTError:
        return None
```

## 9. 优势总结

### 9.1 职责清晰

✅ **Spring Boot**：擅长的业务逻辑、权限控制、数据管理
✅ **Python**：专注 Agent 执行和实时推送
✅ **各司其职**：发挥各自优势

### 9.2 数据准确

✅ **Token 消耗**：Python 执行时直接采集
✅ **执行时长**：Python 精确记录
✅ **配额更新**：Python 调用 Spring Boot API 更新

### 9.3 安全可控

✅ **认证授权**：Spring Boot 统一管理
✅ **Token 共享**：Python 验证 Spring Boot 的 Token
✅ **内部 API**：使用密钥保护

### 9.4 易于扩展

✅ **Spring Boot**：可以轻松添加其他业务功能
✅ **Python**：专注 Agent 执行，不影响业务逻辑
✅ **独立部署**：两个服务可以独立扩展

## 10. 总结

这是**最佳平衡方案**：

- **Spring Boot**：负责用户管理、认证、配额、CRUD（你已经熟悉的）
- **Python Agent**：只负责对话执行和 WebSocket 推送（数据采集准确）
- **Token 共享**：Python 验证 Spring Boot 生成的 Token
- **API 调用**：Python 调用 Spring Boot 内部 API 更新配额

这样既保证了数据采集的准确性，又充分利用了 Spring Boot 的企业级能力！
